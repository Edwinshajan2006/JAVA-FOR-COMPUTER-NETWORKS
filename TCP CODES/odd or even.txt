import java.net.*;
import java.util.*;
import java.util.Scanner;
import java.io.*;

public class Client{
    public static void main (String[] args)throws Exception{
        Socket s =  new Socket("localhost",5000);
        Scanner sc = new Scanner(System.in);
        DataInputStream dis = new DataInputStream(s.getInputStream());
        DataOutputStream dos = new DataOutputStream(s.getOutputStream());

        System.out.println("Enter the number you wanna know is even or not...");
        int a = sc.nextInt();
        dos.writeInt(a);
        String result=dis.readUTF();
        System.out.println("Data received from the Server");
        Thread.sleep(5000);
        System.out.println("Result from the Server"+ result);
        s.close();

    }
}






import java.util.*;
import java.io.*;
import java.util.Scanner;
import java.net.*;


public class Server{
    public static void main (String[] args)throws Exception{
        System.out.println("Waiting for Connection with Client \n");
        ServerSocket ss= new ServerSocket(5000);
        Socket s = ss.accept();
        System.out.println("Successfully Connected with Client...\n");

        DataInputStream dis=new DataInputStream(s.getInputStream());
        DataOutputStream dos=new DataOutputStream(s.getOutputStream());

        int a =dis.readInt();
        String result;

        if (a%2==0){
            result="Even";

        }
        else{
            result="Odd";
        }

        dos.writeUTF(result);

        s.close();
        ss.close();


    }
}
