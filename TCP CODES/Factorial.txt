import java.util.Scanner;
import java.io.*;
import java.net.*;

public class Server{
    public static void main (String[] args)throws Exception{
        ServerSocket ss=new ServerSocket(5000);
        Socket s=ss.accept();
        DataInputStream dis = new DataInputStream(s.getInputStream());
        DataOutputStream dos = new DataOutputStream(s.getOutputStream());
        int n = dis.readInt();
        long fact=1;

        for (int i =1;i<=n;i++){
            fact=fact*i;

        }
        System.out.println(fact + "\n");
        dos.writeLong(fact);
        s.close();
        ss.close();
    }
}






import java.util.Scanner;
import java.io.*;
import java.net.*;

public class Client{
    public static void main(String[] args)throws Exception{
        Socket s=new Socket("localhost",5000);
        Scanner sc=new Scanner(System.in);
        DataInputStream dis = new DataInputStream(s.getInputStream());
        DataOutputStream dos = new DataOutputStream(s.getOutputStream());
        System.out.println("Enter the number you want to find factorial for...\n");
        int n = sc.nextInt();

        dos.writeInt(n);
        long result=dis.readLong();
        System.out.println("Factorial of the give number  : "+ result);
        s.close();

        
    }
}
