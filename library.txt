import java.io.*;
import java.net.*;
import java.util.Scanner;

public class LibraryClient {
    public static void main(String[] args) {
        try (Scanner sc = new Scanner(System.in);
             Socket socket = new Socket("localhost", 5001);
             BufferedReader in = new BufferedReader(
                     new InputStreamReader(socket.getInputStream()));
             PrintWriter out = new PrintWriter(socket.getOutputStream(), true)) {

            System.out.print("Enter book title or book ID: ");
            out.println(sc.nextLine());

            String line;
            while ((line = in.readLine()) != null) {
                System.out.println(line);
            }
        } catch (IOException e) {
            System.out.println("Unable to connect: " + e.getMessage());
        }
    }
}





import java.io.*;
import java.net.*;
import java.util.*;
import java.util.concurrent.*;

public class LibraryServer {
    private static final int PORT = 5001;

    private static final Map<String, String[]> books = new HashMap<>();

    static {
        books.put("101", new String[]{"The Alchemist", "Paulo Coelho", "Available"});
        books.put("102", new String[]{"Wings of Fire", "A. P. J. Abdul Kalam", "Issued"});
        books.put("103", new String[]{"Clean Code", "Robert C. Martin", "Available"});
        books.put("104", new String[]{"The Hobbit", "J. R. R. Tolkien", "Available"});
    }

    public static void main(String[] args) {
        ExecutorService pool = Executors.newCachedThreadPool();
        long pid = ProcessHandle.current().pid();

        try (ServerSocket server = new ServerSocket(PORT)) {
            System.out.println("Library server running on port " + PORT
                    + " | PID: " + pid);

            while (true) {
                Socket socket = server.accept();
                pool.execute(() -> handleClient(socket, pid));
            }
        } catch (IOException e) {
            e.printStackTrace();
        } finally {
            pool.shutdown();
        }
    }

    private static void handleClient(Socket socket, long pid) {
        try (Socket s = socket;
             BufferedReader in = new BufferedReader(
                     new InputStreamReader(s.getInputStream()));
             PrintWriter out = new PrintWriter(s.getOutputStream(), true)) {

            String query = in.readLine();
            String response = "Server PID: " + pid + "\n";

            String[] found = null;
            for (Map.Entry<String, String[]> entry : books.entrySet()) {
                if (entry.getKey().equalsIgnoreCase(query)
                        || entry.getValue()[0].equalsIgnoreCase(query)) {
                    found = entry.getValue();
                    break;
                }
            }

            if (found == null) {
                response += "ERROR: Book not found";
            } else {
                response += "Title: " + found[0] + "\n"
                        + "Author: " + found[1] + "\n"
                        + "Availability: " + found[2];
            }
            out.println(response);
        } catch (IOException e) {
            System.out.println("Client connection error: " + e.getMessage());
        }
    }
}
